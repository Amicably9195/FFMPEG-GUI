# Drawing2CAD runtime dependencies.
#
# IMPORTANT — OpenCV must be the *contrib* build. RapidOCR pulls in plain
# opencv-python as a dependency, which OVERWRITES the contrib build and breaks
# ximgproc (the centerline thinning the vectorizer relies on). So install in
# this order:
#
#     pip install -r requirements.txt
#     pip uninstall -y opencv-python opencv-python-headless
#     pip install --force-reinstall --no-deps opencv-contrib-python-headless
#
# i.e. always let opencv-contrib-python-headless win LAST. (The CI workflow and
# README document the same dance.)

ezdxf>=1.1
pytesseract>=0.3.10

pymupdf>=1.23
rapidocr_onnxruntime>=1.3

# these; the desktop app drawing2cad_gui.py does):
